# M1 Shadow Phase Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Full copybot in `paper` mode — watches the leader, computes the mirrored ladder, gates it, simulates fills, and shows everything on the catseye-style dashboard — running standalone from `start_copybot.bat`.

**Architecture:** One asyncio process. Watcher polls/streams leader state → Sizer computes mirror orders (pure) → Risk gates (pure) → Paper executor simulates our fills whenever the leader fills (same prices ⇒ same fills, × scale). SQLite is the single source of truth; Dash reads it read-only in a daemon thread.

**Tech Stack:** Python 3.10+, `hyperliquid-python-sdk` (info API only in M1 — no signing), `dash`/`plotly`, `pydantic`, `structlog`, `pyyaml`, `pytest`.

## Global Constraints

- Leader address: `0xdae4df7207feb3b350e4284c8efe5f7dac37f637` (config `leader`)
- BTC perp only; sz rounded DOWN to 5 decimals; $10 Hyperliquid min notional
- Mirror parity: our exposure ≤ `leader × scale × 1.05` (PRD §6.1 B2)
- Kill-switch: −35% from high-water mark → HALT (PRD §6.2 M1)
- Scale re-balance only when scale moves > 5%; reconciliation drift trigger 1%
- Paper equity starts at $10,000; `mode: paper` is the config default
- Dashboard port 8061; refresh 15 s; pending-order overlay is an acceptance criterion
- Logging via structlog JSON to `logs/copybot.jsonl`; never `print()`, never f-string log messages, never `event=` kwarg
- All DB writes append-only for `decisions`, `fills`, `events`
- Run tests with `venv\Scripts\python.exe -m pytest` (bare `python` is broken on this machine)

```python
# src/store.py
import sqlite3
from pathlib import Path
from src.models import AccountState

SCHEMA = """
CREATE TABLE IF NOT EXISTS snapshots(
  ts INTEGER NOT NULL, who TEXT NOT NULL, equity REAL, position_btc REAL,
  entry_px REAL, upnl REAL, leverage REAL, mark_px REAL, PRIMARY KEY(ts, who));
CREATE TABLE IF NOT EXISTS decisions(
  id INTEGER PRIMARY KEY, ts INTEGER NOT NULL, trigger TEXT,
  leader_pos REAL, scale REAL, target REAL, delta REAL,
  action TEXT NOT NULL, veto_reason TEXT, risk_state TEXT NOT NULL);
CREATE TABLE IF NOT EXISTS orders(
  oid INTEGER PRIMARY KEY, decision_id INTEGER, ts INTEGER, side TEXT,
  px REAL, sz REAL, exec_style TEXT, status TEXT,
  filled_sz REAL DEFAULT 0, avg_px REAL, fees REAL);
CREATE TABLE IF NOT EXISTS fills(
  tid INTEGER PRIMARY KEY, oid INTEGER, ts INTEGER, side TEXT, px REAL,
  sz REAL, crossed INTEGER, closed_pnl REAL, fee REAL);
CREATE TABLE IF NOT EXISTS leader_fills(
  tid INTEGER PRIMARY KEY, ts INTEGER, side TEXT, px REAL, sz REAL,
  crossed INTEGER, dir TEXT);
CREATE TABLE IF NOT EXISTS leader_open_orders(
  snapshot_ts INTEGER, oid INTEGER, side TEXT, px REAL, sz REAL,
  PRIMARY KEY(snapshot_ts, oid));
CREATE TABLE IF NOT EXISTS mirror_map(
  leader_oid INTEGER PRIMARY KEY, our_oid INTEGER, px REAL,
  leader_sz REAL, our_sz REAL, scale_used REAL,
  created_ts INTEGER, closed_ts INTEGER, close_reason TEXT);
CREATE TABLE IF NOT EXISTS equity_curve(
  ts INTEGER PRIMARY KEY, equity REAL, hwm REAL, drawdown_pct REAL,
  funding_cum REAL DEFAULT 0, fees_cum REAL DEFAULT 0, realized_cum REAL DEFAULT 0);
CREATE TABLE IF NOT EXISTS events(
  id INTEGER PRIMARY KEY, ts INTEGER, level TEXT, kind TEXT, message TEXT);
"""

class Store:
    def __init__(self, db_path: str | Path):
        Path(db_path).parent.mkdir(parents=True, exist_ok=True)
        self.conn = sqlite3.connect(str(db_path), timeout=30)
        self.conn.execute("PRAGMA journal_mode=WAL")
        self.conn.executescript(SCHEMA)
        self.conn.commit()

    def record_snapshot(self, who: str, s: AccountState) -> None:
        upnl = (s.mark_px - s.entry_px) * s.position if s.entry_px else 0.0
        lev = abs(s.position) * s.mark_px / s.equity if s.equity else 0.0
        self.conn.execute(
            "INSERT OR REPLACE INTO snapshots VALUES (?,?,?,?,?,?,?,?)",
            (s.fetched_at_ms, who, s.equity, s.position, s.entry_px, upnl, lev, s.mark_px))
        self.conn.commit()

    def record_decision(self, ts_ms, trigger, action, risk_state, veto_reason="",
                        leader_pos=None, scale=None, target=None, delta=None) -> int:
        cur = self.conn.execute(
            "INSERT INTO decisions(ts,trigger,leader_pos,scale,target,delta,action,veto_reason,risk_state)"
            " VALUES (?,?,?,?,?,?,?,?,?)",
            (ts_ms, trigger, leader_pos, scale, target, delta, action, veto_reason, risk_state))
        self.conn.commit()
        return cur.lastrowid

    def record_event(self, ts_ms, level, kind, message) -> None:
        self.conn.execute("INSERT INTO events(ts,level,kind,message) VALUES (?,?,?,?)",
                          (ts_ms, level, kind, message))
        self.conn.commit()

    def mirror_get(self) -> dict[int, dict]:
        rows = self.conn.execute(
            "SELECT leader_oid, our_oid, px, leader_sz, our_sz, scale_used "
            "FROM mirror_map WHERE closed_ts IS NULL").fetchall()
        keys = ["leader_oid", "our_oid", "px", "leader_sz", "our_sz", "scale_used"]
        return {r[0]: dict(zip(keys, r)) for r in rows}

    def mirror_put(self, leader_oid, our_oid, px, leader_sz, our_sz, scale, ts_ms):
        self.conn.execute(
            "INSERT OR REPLACE INTO mirror_map VALUES (?,?,?,?,?,?,?,NULL,NULL)",
            (leader_oid, our_oid, px, leader_sz, our_sz, scale, ts_ms))
        self.conn.commit()

    def mirror_close(self, leader_oid, ts_ms, reason):
        self.conn.execute(
            "UPDATE mirror_map SET closed_ts=?, close_reason=? WHERE leader_oid=?",
            (ts_ms, reason, leader_oid))
        self.conn.commit()

    def update_equity(self, ts_ms, equity) -> float:
        prev = self.conn.execute("SELECT MAX(hwm) FROM equity_curve").fetchone()[0]
        hwm = max(prev or 0.0, equity)
        dd = (equity / hwm - 1) * 100 if hwm else 0.0
        self.conn.execute(
            "INSERT OR REPLACE INTO equity_curve(ts,equity,hwm,drawdown_pct) VALUES (?,?,?,?)",
            (ts_ms, equity, hwm, dd))
        self.conn.commit()
        return dd

    def latest_risk_state(self) -> str:
        row = self.conn.execute(
            "SELECT message FROM events WHERE kind='state_change' ORDER BY id DESC LIMIT 1"
        ).fetchone()
        return row[0] if row else "NORMAL"
```

```python
# src/sizer.py
import math
from src.models import Order, MirrorAction

SZ_DECIMALS = 5

def compute_scale(our_equity: float, leader_equity: float) -> float:
    return our_equity / leader_equity

def mirror_size(leader_sz: float, scale: float, px: float,
                min_notional_usd: float = 10.0) -> float:
    sz = math.floor(leader_sz * scale * 10**SZ_DECIMALS) / 10**SZ_DECIMALS
    return sz if sz * px >= min_notional_usd else 0.0

def diff_ladders(leader_orders: list[Order], mirror: dict[int, dict],
                 scale: float, rebalance_pct: float) -> list[MirrorAction]:
    actions: list[MirrorAction] = []
    his = {o.oid: o for o in leader_orders}
    for loid, m in mirror.items():
        stale_scale = abs(scale / m["scale_used"] - 1) * 100 > rebalance_pct
        if loid not in his or stale_scale:
            actions.append(MirrorAction(kind="cancel", side="B", px=m["px"],
                                        sz=m["our_sz"], leader_oid=loid,
                                        our_oid=m["our_oid"]))
            if stale_scale and loid in his:      # re-place at new scale
                o = his[loid]
                sz = mirror_size(o.sz, scale, o.px)
                if sz:
                    actions.append(MirrorAction(kind="place", side=o.side,
                                                px=o.px, sz=sz, leader_oid=loid))
    for oid, o in his.items():
        if oid not in mirror:
            sz = mirror_size(o.sz, scale, o.px)
            if sz:
                actions.append(MirrorAction(kind="place", side=o.side,
                                            px=o.px, sz=sz, leader_oid=oid))
    return actions

def position_delta(leader_pos: float, scale: float, our_pos: float,
                   drift_pct: float) -> float:
    target = leader_pos * scale
    if target and abs(our_pos / target - 1) * 100 < drift_pct:
        return 0.0
    delta = target - our_pos
    return math.copysign(
        math.floor(abs(delta) * 10**SZ_DECIMALS) / 10**SZ_DECIMALS, delta)
```

```python
# src/risk.py
from src.config import Config
from src.models import AccountState, MirrorAction, RiskState, Verdict

def _exposure(a: AccountState) -> float:
    resting = sum(o.sz * o.px for o in a.open_orders)
    return abs(a.position) * a.mark_px + resting

def check_order(a: MirrorAction, ours: AccountState, leader: AccountState,
                now_ms: int, state: RiskState, cfg: Config) -> Verdict:
    if state == RiskState.HALT:
        return Verdict(approved=False, reason="B5_state")
    if state == RiskState.WARNING and a.kind == "place":
        return Verdict(approved=False, reason="B5_state")
    if (now_ms - leader.fetched_at_ms) / 1000 > cfg.risk.leader_staleness_max_s:
        return Verdict(approved=False, reason="B4_stale")
    if a.kind == "place":
        scale = ours.equity / leader.equity
        cap = _exposure(leader) * scale * cfg.risk.mirror_parity_tolerance
        if _exposure(ours) + a.sz * a.px > cap:
            return Verdict(approved=False, reason="B2_parity")
    return Verdict(approved=True)

def run_monitors(drawdown_pct: float, leader_age_s: float,
                 state: RiskState, cfg: Config) -> tuple[RiskState, list[str]]:
    alerts: list[str] = []
    if state == RiskState.HALT:
        return RiskState.HALT, alerts
    if drawdown_pct <= cfg.risk.max_drawdown_pct:
        alerts.append(f"kill_switch drawdown {drawdown_pct:.1f}%")
        return RiskState.HALT, alerts
    if leader_age_s > cfg.risk.leader_staleness_max_s:
        if state != RiskState.WARNING:
            alerts.append(f"leader_stale {leader_age_s:.0f}s")
        return RiskState.WARNING, alerts
    return RiskState.NORMAL, alerts
```

Note: B1 (BTC-only) and B3 (price integrity) are enforced structurally in M1 — the watcher only parses BTC and mirror orders always carry his price — so `check_order` covers B2/B4/B5. B1/B3 become real checks in M2 when the live executor exists.

```python
import aiohttp
from src.models import AccountState, Order

def parse_clearinghouse(raw: dict, orders_raw: list[dict],
                        mark_px: float, now_ms: int) -> AccountState:
    pos, entry = 0.0, None
    for ap in raw.get("assetPositions", []):
        p = ap["position"]
        if p["coin"] == "BTC":                       # B1: BTC only, by construction
            pos = float(p["szi"])
            entry = float(p["entryPx"]) if p.get("entryPx") else None
    orders = [Order(oid=o["oid"], side=o["side"], px=float(o["limitPx"]),
                    sz=float(o["sz"]), ts_ms=o["timestamp"])
              for o in orders_raw if o.get("coin") == "BTC"]
    return AccountState(
        equity=float(raw["marginSummary"]["accountValue"]),
        position=pos, entry_px=entry, mark_px=mark_px,
        fetched_at_ms=now_ms, open_orders=orders)

class Watcher:
    def __init__(self, address: str, api_url: str = "https://api.hyperliquid.xyz"):
        self.address, self.api_url = address, api_url

    async def _info(self, session, body: dict):
        async with session.post(f"{self.api_url}/info", json=body) as r:
            r.raise_for_status()
            return await r.json()

    async def fetch(self, now_ms: int) -> AccountState:
        async with aiohttp.ClientSession() as s:
            ch = await self._info(s, {"type": "clearinghouseState", "user": self.address})
            oo = await self._info(s, {"type": "frontendOpenOrders", "user": self.address})
            mids = await self._info(s, {"type": "allMids"})
        return parse_clearinghouse(ch, oo, float(mids["BTC"]), now_ms)
```

```python
# src/paper.py
from src.models import AccountState, MirrorAction, Order
from src.store import Store

MAKER_FEE = 0.00015          # 0.015% Hyperliquid maker

class PaperBroker:
    """Simulated account. Same-price rule: our mirrored order fills when the
    leader's does, proportionally, at his price. ponytail: no partial-queue
    modeling — his fill ratio IS our fill ratio."""

    def __init__(self, store: Store, start_equity: float):
        self.store = store
        self.cash = start_equity
        self.position = 0.0
        self.entry_notional = 0.0
        self.open: dict[int, Order] = {}
        self._next_oid = -1                      # synthetic negative oids

    def execute(self, a: MirrorAction, now_ms: int) -> int:
        if a.kind == "cancel":
            self.open.pop(a.our_oid, None)
            return a.our_oid or 0
        oid, self._next_oid = self._next_oid, self._next_oid - 1
        self.open[oid] = Order(oid=oid, side=a.side, px=a.px, sz=a.sz, ts_ms=now_ms)
        return oid

    def on_leader_fill(self, leader_oid: int, fill_sz: float, px: float, now_ms: int):
        m = self.store.mirror_get().get(leader_oid)
        if not m or m["our_oid"] not in self.open:
            return
        o = self.open[m["our_oid"]]
        ratio = min(1.0, fill_sz / m["leader_sz"])
        sz = round(o.sz * ratio, 5)
        signed = sz if o.side == "B" else -sz
        self.position += signed
        self.entry_notional += signed * px
        self.cash -= sz * px * MAKER_FEE
        remaining = round(o.sz - sz, 5)
        if remaining > 0:
            self.open[o.oid] = o.model_copy(update={"sz": remaining})
        else:
            del self.open[o.oid]
        tid = -now_ms
        self.store.conn.execute(
            "INSERT OR REPLACE INTO fills VALUES (?,?,?,?,?,?,?,?,?)",
            (tid, o.oid, now_ms, o.side, px, sz, 0, 0.0, sz * px * MAKER_FEE))
        self.store.conn.commit()

    def state(self, mark_px: float, now_ms: int) -> AccountState:
        upnl = self.position * mark_px - self.entry_notional
        entry = (self.entry_notional / self.position) if self.position else None
        return AccountState(equity=self.cash + upnl, position=self.position,
                            entry_px=entry, mark_px=mark_px, fetched_at_ms=now_ms,
                            open_orders=list(self.open.values()))
```

```python
# src/main.py
import asyncio, os, sys, threading, time, webbrowser
from dataclasses import dataclass
from pathlib import Path
import structlog
from src.config import Config, load_config
from src.models import RiskState
from src.paper import PaperBroker
from src.risk import check_order, run_monitors
from src.sizer import compute_scale, diff_ladders
from src.store import Store
from src.watcher import Watcher

log = structlog.get_logger(__name__)

@dataclass
class Deps:
    cfg: Config
    store: Store
    watcher: object
    broker: PaperBroker

async def cycle(deps: Deps, now_ms: int) -> None:
    leader = await deps.watcher.fetch(now_ms)
    ours = deps.broker.state(leader.mark_px, now_ms)
    deps.store.record_snapshot("leader", leader)
    deps.store.record_snapshot("copy", ours)
    for o in leader.open_orders:
        deps.store.conn.execute(
            "INSERT OR REPLACE INTO leader_open_orders VALUES (?,?,?,?,?)",
            (now_ms, o.oid, o.side, o.px, o.sz))
    deps.store.conn.commit()

    dd = deps.store.update_equity(now_ms, ours.equity)
    state = RiskState(deps.store.latest_risk_state())
    age_s = (now_ms - leader.fetched_at_ms) / 1000
    new_state, alerts = run_monitors(dd, age_s, state, deps.cfg)
    if new_state != state:
        deps.store.record_event(now_ms, "critical" if new_state == RiskState.HALT
                                else "warning", "state_change", new_state.value)
        for a in alerts:
            deps.store.record_event(now_ms, "critical", "monitor", a)
    state = new_state

    scale = compute_scale(ours.equity, leader.equity)
    actions = diff_ladders(leader.open_orders, deps.store.mirror_get(),
                           scale, deps.cfg.mirror.scale_rebalance_pct)
    for a in actions:
        v = check_order(a, ours, leader, now_ms, state, deps.cfg)
        deps.store.record_decision(
            now_ms, "poll", "order" if v.approved else "veto",
            state.value, veto_reason=v.reason,
            leader_pos=leader.position, scale=scale)
        if not v.approved:
            continue
        our_oid = deps.broker.execute(a, now_ms)
        if a.kind == "place":
            deps.store.mirror_put(a.leader_oid, our_oid, a.px,
                                  a.sz / scale, a.sz, scale, now_ms)
        else:
            deps.store.mirror_close(a.leader_oid, now_ms, "his_cancel")
    if not actions:
        deps.store.record_decision(now_ms, "poll", "skip_dust", state.value)

async def run(cfg: Config) -> None:
    store = Store(cfg.storage.db_path)
    deps = Deps(cfg=cfg, store=store, watcher=Watcher(cfg.leader),
                broker=PaperBroker(store, cfg.paper.start_equity))
    while True:
        try:
            await cycle(deps, int(time.time() * 1000))
        except Exception:
            log.exception("cycle_failed")
        await asyncio.sleep(cfg.storage.snapshot_interval_s)

def main() -> None:
    cfg = load_config("config.yaml")
    pid = Path("logs/copybot.pid")
    pid.parent.mkdir(exist_ok=True)
    pid.write_text(str(os.getpid()))
    try:
        from src.dashboard import make_app
        app = make_app(cfg)
        threading.Thread(target=lambda: app.run(port=cfg.dashboard.port,
                                                debug=False), daemon=True).start()
        asyncio.run(run(cfg))
    finally:
        pid.unlink(missing_ok=True)

if __name__ == "__main__":
    main()
```
